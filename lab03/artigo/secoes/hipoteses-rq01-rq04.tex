\begin{itemize}
\item \textbf{RQ 01 (frequência de deploy):} a faixa \textit{Medium} deve ser a mais comum e a \textit{Elite}, minoria, porque muitos projetos fazem releases manuais ou por marcos, e estrelas indicam visibilidade, não cadência; a \textit{Low} deve reunir os repositórios com 5 a 11 releases no ano.
\item \textbf{RQ 02 (lead time):} de dias a poucas semanas, na maioria \textit{Medium} e \textit{High}, com a variante (a) maior e com mais repositórios \textit{Low} que a (b), já que basta um commit antigo ``esquecido'' para elevá-la.
\item \textbf{RQ 03 (CFR):} baixo na maioria dos repositórios (\textit{Elite} ou \textit{High}), com grande dispersão, e a variante (a) maior que a (b), por incluir testes instáveis e falhas de infraestrutura que não chegam ao usuário.
\item \textbf{RQ 04 (tempo de recuperação):} menos de um dia, com mediana \textit{High}, poucos repositórios \textit{Elite}, IQR amplo, cauda de dias a semanas e uma parcela de episódios censurados.
\end{itemize}
